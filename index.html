function handleAct(a,sp){

  if(G.phase!=='GAME'||G.players[G.act].peerId!==sp) return;

  if(a.t==='ROLL') doRoll();

  else if(a.t==='QUICK_REROLL') quickRerollSingleDie(a.i);

  else if(a.t==='RESOLVE') startResolve();

  else if(a.t==='CANCEL_RESOLVE') cancelResolve();

  else if(a.t==='TARGET') applyTarget(a.id,a.dbl);

  else if(a.t==='TARGET_KIT') applyKitArrowTarget(a.id);

  else if(a.t==='BART_CHOICE') applyBartChoice(a.takeArrow);

  else if(a.t==='SKIP_BEER'){ G.pending.beer=0; resolveStep(); }

  else if(a.t==='SKIP_KIT'){ G.pending.kitArrows=0; resolveStep(); }

  else if(a.t==='FIRE_GATLING') fireGatling();

}

function doRoll(){

  const cp=G.players[G.act]; if(!cp||!cp.alive){ endTurn(); return; }

  document.querySelectorAll('.die').forEach(d=>d.classList.add('rolling'));

  setTimeout(()=>document.querySelectorAll('.die').forEach(d=>d.classList.remove('rolling')),360);

    for(let d of G.dice){

      d.face=F[Math.floor(Math.random()*F.length)]; d.rerolls=0; d.sel=false;

      if(d.face.id==='arrow') takeArrow(G.act);

      else if(d.face.id==='dynamite'&&cp.character.n!=="Black Jack") d.locked=true;

      if(checkEnd('🏹 Az Indiántámadás végzett a Seriffel!')) return;

      if(!cp.alive){ if(mode==='LOCAL')sync(); else bcast(); endTurn(); return; }

    }

  } else {

    if(G.rule){

      const toR=G.dice.filter(d=>d.sel&&!d.locked&&d.rerolls<2);

      if(!toR.length||toR.length>G.budget) return;

      G.budget-=toR.length;

      for(let d of toR){

        d.face=F[Math.floor(Math.random()*F.length)]; d.rerolls++; d.sel=false;

        if(d.face.id==='arrow') takeArrow(G.act);

        else if(d.face.id==='dynamite'&&cp.character.n!=="Black Jack") d.locked=true;

        if(checkEnd('🏹 Az Indiántámadás végzett a Seriffel!')) return;

        if(!cp.alive){ if(mode==='LOCAL')sync(); else bcast(); endTurn(); return; }

      }

      if(G.rolls>=G.maxRolls) return;

      }

    }

function resolveStep(){

  if(checkEnd()) return;

  const cp=G.players[G.act]; if(!cp||!cp.alive){ G.actType=null; endTurn(); return; }

  if(G.pending.kitArrows>0) G.actType='kitArrows';

  else if(G.pending.s1>0) G.actType='shoot1';

  else if(G.pending.s2>0) G.actType='shoot2';

  else if(G.pending.beer>0) G.actType='beer';

  else { G.actType=null; endTurn(); return; }

  if(G.actType.startsWith('shoot')){

    const dist=G.actType==='shoot1'?1:2, tgts=getTargets(dist);

    if(!tgts.length){

      log(`🎯${dist} Nincs elérhető célpont ezen a távon!`,'event');

      if(G.actType==='shoot1') G.pending.s1=0; else G.pending.s2=0;

      resolveStep(); return;

    }

  }

  if(G.actType==='kitArrows'&&!G.players.some(p=>p.alive&&p.arrows>0)){

    G.pending.kitArrows=0; resolveStep(); return;

  const a=cp.name, t=G.players[tIdx].name;

function botTurn(tId){

  if(tId!==currentTurnId||!G||G.phase!=='GAME') return;

  const cp=G.players[G.act]; if(!cp||!cp.isBot||!cp.alive) return;

  doRoll();

  botTimer=setTimeout(()=>{

    if(tId!==currentTurnId||!G||G.phase!=='GAME') return;

    const curr=G.players[G.act]; if(!curr||!curr.isBot||!curr.alive) return;

    if(G.actType==='askGatling'||G.actType==='forceGatling'){ fireGatling(); return; }

    if(G.actType) return;

    const sheriff=G.players.find(p=>p.role==='Seriff');

    const wantBeer=curr.hp<=3||(curr.role==='Helyettes'&&sheriff&&sheriff.hp<=4);

    let s=0;

    G.dice.forEach(d=>{

      if(d.locked||(G.rule&&d.rerolls>=2)) return;

      const keep=(d.face.id==='beer'&&wantBeer)||(d.face.id==='gatling')||(d.face.id.startsWith('shoot')&&curr.role==='Bandita');

      const maxS=G.rule?G.budget:5;

      if(!keep&&s<maxS){

        const spent=G.maxBudget-G.budget;

        if(G.rule&&spent+s+1>=3&&d.rerolls>0){} else { d.sel=true; s++; }

      }

    });

    if(s>0){ doRoll(); botTimer=setTimeout(()=>{ if(tId===currentTurnId) startResolve(); },750); }

    else startResolve();

  },750);

}

function botAction(tId){

  if(tId!==currentTurnId||!G||G.phase!=='GAME') return;

  const cp=G.players[G.act]; if(!cp||!cp.isBot||!cp.alive||!G.actType) return;

  const sheriff=G.players.find(p=>p.role==='Seriff'), liv=G.players.filter(p=>p.alive);

  if(G.actType==='kitArrows'){

    let target=G.players.find(p=>p.alive&&p.arrows>0&&p.id===cp.id)||

               G.players.find(p=>p.alive&&p.arrows>0&&(cp.role==='Helyettes'?p.role==='Seriff':true))||

               G.players.find(p=>p.alive&&p.arrows>0);

    if(target) applyKitArrowTarget(target.id); else { G.pending.kitArrows=0; resolveStep(); }

  } else if(G.actType.startsWith('shoot')){

    const dist=G.actType==='shoot1'?1:2, tgts=getTargets(dist);

    if(!tgts.length){ if(G.actType==='shoot1') G.pending.s1=0; else G.pending.s2=0; resolveStep(); return; }

    let ch=tgts[0];

    if(cp.role==='Bandita') ch=tgts.find(t=>t.role==='Seriff')||tgts[0];

    else if(cp.role==='Helyettes') ch=tgts.find(t=>t.role!=='Seriff')||tgts[0];

    else if(cp.role==='Renegát') ch=(liv.length===2&&tgts.find(t=>t.role==='Seriff'))?sheriff:(tgts.find(t=>t.role!=='Seriff')||tgts[0]);

    else if(cp.role==='Seriff') ch=tgts[Math.floor(Math.random()*tgts.length)];

    const useDbl=(cp.character.n==="Slab the Killer"&&G.pending.beer>0&&(ch.role==='Seriff'||ch.hp<=2));

    applyTarget(ch.id,useDbl);

  } else if(G.actType==='beer'){

    if(cp.role==='Helyettes'&&sheriff&&sheriff.alive&&sheriff.hp<sheriff.maxHp) applyTarget(sheriff.id,false);

    else if(cp.hp<cp.maxHp) applyTarget(cp.id,false);

    else { G.pending.beer=0; resolveStep(); }

  }

}

function getTargets(dist){

  const circle=G.players.filter(p=>p.alive), cLen=circle.length;

  if(cLen<=1) return [];

  const cp=G.players[G.act], cIdx=circle.findIndex(p=>p.id===cp.id);

  if(cIdx===-1) return [];

  let candidates=[];

  if(cp.character.n==="Calamity Janet"||cp.character.n==="Rose Doolan"){

    const r1=circle[(cIdx+1)%cLen], l1=circle[(cIdx-1+cLen)%cLen], r2=circle[(cIdx+2)%cLen], l2=circle[(cIdx-2+cLen)%cLen], r3=circle[(cIdx+3)%cLen], l3=circle[(cIdx-3+cLen)%cLen];

    if(cp.character.n==="Calamity Janet") candidates=[r1,l1,r2,l2];

    else if(dist===1) candidates=[r1,l1,r2,l2];

    else if(dist===2) candidates=[r2,l2,r3,l3];

  } else if(dist===1){

    candidates=(cLen===2)?[circle[(cIdx+1)%cLen]]:[circle[(cIdx+1)%cLen],circle[(cIdx-1+cLen)%cLen]];

  } else if(dist===2){

    candidates=(cLen<=3)?getTargets(1):[circle[(cIdx+2)%cLen],circle[(cIdx-2+cLen)%cLen]];

  }

  return [...new Set(candidates)].filter(p=>p&&p.id!==cp.id);

}
